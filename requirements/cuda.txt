### NVIDIA/CUDA GPU Requirements ###
# For systems with NVIDIA GPUs and CUDA installed

# Step 1: Install base requirements first
-r base.txt

# Step 2: Install PyTorch with CUDA support
# IMPORTANT: Don't install these directly! Use the command below instead:
#
# For CUDA 12.1:
#   pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121
#
# For CUDA 11.8:
#   pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu118
#
# Choose the version matching your CUDA toolkit
# See: https://pytorch.org/get-started/locally/

# Step 3: (Optional) Install PyTorch-specific extras
# Uncomment if you want quantization support:
# accelerate>=0.27.0
# bitsandbytes>=0.41.3
